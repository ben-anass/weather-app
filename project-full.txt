import { WEATHER_CODES } from "./weatherCodes.js";

export function displayWeather(cityData, weatherData) {
    const container = document.querySelector("#weatherContainer")

    container.textContent = "" ;
    const current = weatherData.current;
    
    const weatherCode = current.weather_code 

    const weatherInfo = WEATHER_CODES[weatherCode] || {
        label : "Conditions meteo inconues", icon : "?"}
    
    const temperature = current.temperature_2m 
    const humidity = current.relative_humidity_2m 
    const windSpeed = current.wind_speed_10m 

    const card = document.createElement("article");
    card.classList.add("weather-card")

    const cityName = document.createElement("h2")
    cityName.textContent = cityData.name 

    const icon = document.createElement("p")
    icon.textContent = weatherInfo.icon 
    icon.classList.add("weather-icon");

    const description = document.createElement("p")
    description.textContent = weatherInfo.label 


    const temp = document.createElement("p")
    temp.textContent = `${temperature} °C` ;

    const humidityText = document.createElement("p")
    humidityText.textContent =`Humidite : ${humidity} %` ;

    const wind = document.createElement("p")
    wind.textContent = ` vent : ${windSpeed} km/h`;

    card.append(cityName , icon , description , temp , humidityText , wind  ) 

    container.appendChild(card);

}

export function createForecastCard(forecast) {
    const card = document.createElement("article")
    card.classList.add("forecast-card") ;

    const date = document.createElement("h3")
    const formattedDate = new Date(forecast.date + "T00:00:00")

    date.textContent = formattedDate.toLocaleDateString("fr-FR", {
        weekday : "long" , day : "numeric" , month : "long" 
    })

    const maxTemp = document.createElement("p") 
    maxTemp.textContent = `Max : ${forecast.max ?? "Donnée indisponible"} °C`

    const minTemp = document.createElement("p")
    minTemp.textContent = `Min : ${forecast.min ?? "Donnée indisponible"} °C` 
    
    card.append(date , maxTemp  , minTemp )


    return  card ;
}



export function displayForecasts(forecasts) {
    const container = document.querySelector("#forecastList")
    container.textContent = "" 

    const cards = forecasts.map(function(forecast) {
        return createForecastCard(forecast ) ;
    })
    container.append(...cards)
}

export async function getCoordinates(city) {
    const url = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1&language=fr&format=json`;

    const response = await fetch(url)
    if (!response.ok) {
        throw new Error("Entre lors de la rechrche de la ville ")
    }
    const data = await response.json() ;

    if (!data.results || data.results.length === 0) {
        throw new Error("Ville introuvable.");
    }
    return data.results[0]
}


export async function getCurrentWeather(latitude, longitude) {
    
    const url = `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m&daily=temperature_2m_max,temperature_2m_min,weather_code&forecast_days=5&temperature_unit=celsius&wind_speed_unit=kmh&timezone=auto`;

    const response = await fetch(url);

    if (!response.ok) {
        if (response.status === 429) {
            throw new Error("Quota atteint. Veuillez réessayer plus tard.")
        }
        throw new Error("Le service météo est temporairement indisponible.")
    }
    
    const data = await response.json();

    return data;
}

========== ./js/app.js ==========
import { getCoordinates, getCurrentWeather } from "./api.js";
import { displayWeather , displayForecasts } from "./ui.js";


const searchForm = document.querySelector("#searchForm");
const cityInput = document.querySelector("#cityInput");
const message = document.querySelector("#message");
const loadingMessage = document.querySelector("#loadingMessage");
const retryButton = document.querySelector("#retryButton");


let lastCity = "";


searchForm.addEventListener("submit", function (event) {
    event.preventDefault()

    const city = cityInput.value.trim();

    if (city === "") {
        message.textContent = "Veuillez saisir une ville."  ;
        retryButton.hidden = true
        return
    }

    lastCity = city
    searchWeather(city);
})


retryButton.addEventListener("click", function () {
    if (lastCity !== ""){
        searchWeather(lastCity) ;
    }
})



async function searchWeather(city) {

    message.textContent = "" 
    loadingMessage.hidden = false
    retryButton.hidden = true;

    try {
        // Recuperer  coordonnees ville
        const cityData = await getCoordinates(city) ;

        // Recuperer  météo 
        const weatherData = await getCurrentWeather(
            cityData.latitude,
            cityData.longitude
        );

        const daily = weatherData.daily ;
        const forecasts = daily.time.map(function(date, index) {
            return {
                date : date , 
                max : daily.temperature_2m_max[index],
                min : daily.temperature_2m_min[index],
                code : daily.weather_code[index]
            } ;
        }) ;


        // Affiche meteo 
        displayWeather(cityData, weatherData)
        displayForecasts(forecasts)

    } catch (error) {
        message.textContent = error.message
        retryButton.hidden = false;

    } finally {
        loadingMessage.hidden = true ;
    }

}
